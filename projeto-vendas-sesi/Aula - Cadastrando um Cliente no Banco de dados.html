Cadastro de Clientes

Java Swing + MySQL usando só métodos, sem classes de modelo

🎯 Sobre esta aula
Todo o código fica dentro da própria classe da tela (Frmclientes.java). Não vamos criar classes de Model, DAO ou Controller — cada botão chama diretamente um método que conversa com o banco. É a forma mais direta de entender como Java conversa com MySQL antes de aprender orientação a objeto de verdade.

2 Criando o banco de dados

No MySQL Workbench (ou phpMyAdmin), crie o banco e a tabela tb_clientes:

5 Criando o método de conexão

Antes de qualquer botão, criamos um método que devolve uma Connection com o banco. Ele fica dentro da própria classe Frmclientes, sem separar em outro arquivo:

Frmclientes.javaprivate Connection conectar() {
    Connection conexao = null;
    try {
        String url = "jdbc:mysql://localhost:3306/db_vendas_sesi";
        String usuario = "root";
        String senha = ""; // coloque a senha do seu MySQL

        conexao = DriverManager.getConnection(url, usuario, senha);
    } catch (SQLException erro) {
        JOptionPane.showMessageDialog(null, "Erro ao conectar: " + erro.getMessage());
    }
    return conexao;
}
💡 Imports necessários
Lá no topo do arquivo, adicione: java.sql.Connection, java.sql.DriverManager, java.sql.PreparedStatement, java.sql.ResultSet, java.sql.SQLException e javax.swing.table.DefaultTableModel.

7 Botão "Salvar" — INSERT

Insere um novo cliente usando PreparedStatement (evita SQL Injection e lida melhor com acentos):

Frmclientes.javaprivate void btnSalvarActionPerformed(java.awt.event.ActionEvent evt) {
    Connection conexao = conectar();
    if (conexao == null) return;

    try {
        String sql = "INSERT INTO tb_clientes (NOME, RG, CPF, EMAIL, TELEFONE, "
                + "CELULAR, CEP, ENDERECO, NUMERO, COMPLEMENTO, BAIRRO, CIDADE, estado) "
                + "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)";

        PreparedStatement stmt = conexao.prepareStatement(sql);
        stmt.setString(1, txtNome.getText());
        stmt.setString(2, txtRG.getText());
        stmt.setString(3, txtCPF.getText());
        stmt.setString(4, txtEmail.getText());
        stmt.setString(5, txtTelefone.getText());
        stmt.setString(6, txtCelular.getText());
        stmt.setString(7, txtCep.getText());
        stmt.setString(8, txtEndereco.getText());
        stmt.setInt(9, Integer.parseInt(txtNumero.getText()));
        stmt.setString(10, txtComplemento.getText());
        stmt.setString(11, txtBairro.getText());
        stmt.setString(12, txtCidade.getText());
        stmt.setString(13, cbUF.getSelectedItem().toString());

        stmt.executeUpdate();
        JOptionPane.showMessageDialog(null, "Cliente salvo com sucesso!");
        conexao.close();
    } catch (SQLException erro) {
        JOptionPane.showMessageDialog(null, "Erro ao salvar: " + erro.getMessage());
    }
}
⚠️ Erro comum
Se o campo N° estiver vazio, Integer.parseInt() quebra o programa. Um jeito simples de tratar: txtNumero.getText().isEmpty() ? 0 : Integer.parseInt(txtNumero.getText()).

8 Aba "Consulta de clientes" — botão Pesquisar

Esse botão busca por nome e preenche a JTable com os resultados:

Frmclientes.javaprivate void btnPesquisarActionPerformed(java.awt.event.ActionEvent evt) {
    Connection conexao = conectar();
    if (conexao == null) return;

    try {
        String sql = "SELECT ID, NOME, RG, CPF, EMAIL, TELEFONE, CELULAR, CEP, "
                + "ENDERECO, NUMERO, COMPLEMENTO, BAIRRO, CIDADE, estado "
                + "FROM tb_clientes WHERE NOME LIKE ?";

        PreparedStatement stmt = conexao.prepareStatement(sql);
        stmt.setString(1, "%" + txtPesquisaNome.getText() + "%");
        ResultSet rs = stmt.executeQuery();

        DefaultTableModel modelo = new DefaultTableModel();
        modelo.addColumn("Código");
        modelo.addColumn("Nome");
        modelo.addColumn("RG");
        modelo.addColumn("CPF");
        modelo.addColumn("E-mail");
        modelo.addColumn("Telefone");
        modelo.addColumn("Celular");
        modelo.addColumn("Cep");
        modelo.addColumn("Endereço");
        modelo.addColumn("N°");
        modelo.addColumn("Comp");
        modelo.addColumn("Bairro");
        modelo.addColumn("Cidade");
        modelo.addColumn("UF");

        while (rs.next()) {
            Object[] linha = {
                rs.getInt("ID"), rs.getString("NOME"), rs.getString("RG"),
                rs.getString("CPF"), rs.getString("EMAIL"), rs.getString("TELEFONE"),
                rs.getString("CELULAR"), rs.getString("CEP"), rs.getString("ENDERECO"),
                rs.getInt("NUMERO"), rs.getString("COMPLEMENTO"), rs.getString("BAIRRO"),
                rs.getString("CIDADE"), rs.getString("estado")
            };
            modelo.addRow(linha);
        }

        tabelaClientes.setModel(modelo);
        conexao.close();
    } catch (SQLException erro) {
        JOptionPane.showMessageDialog(null, "Erro ao pesquisar: " + erro.getMessage());
    }
}

10 Botão "Editar" — UPDATE

Atualiza o cliente que está com o Código preenchido (por isso o campo Código fica desabilitado — ele só é preenchido ao carregar um cliente existente):

Frmclientes.javaprivate void btnEditarActionPerformed(java.awt.event.ActionEvent evt) {
    if (txtCodigo.getText().isEmpty()) {
        JOptionPane.showMessageDialog(null, "Selecione um cliente na Consulta antes de editar.");
        return;
    }

    Connection conexao = conectar();
    if (conexao == null) return;

    try {
        String sql = "UPDATE tb_clientes SET NOME=?, RG=?, CPF=?, EMAIL=?, TELEFONE=?, "
                + "CELULAR=?, CEP=?, ENDERECO=?, NUMERO=?, COMPLEMENTO=?, BAIRRO=?, "
                + "CIDADE=?, estado=? WHERE ID=?";

        PreparedStatement stmt = conexao.prepareStatement(sql);
        stmt.setString(1, txtNome.getText());
        stmt.setString(2, txtRG.getText());
        stmt.setString(3, txtCPF.getText());
        stmt.setString(4, txtEmail.getText());
        stmt.setString(5, txtTelefone.getText());
        stmt.setString(6, txtCelular.getText());
        stmt.setString(7, txtCep.getText());
        stmt.setString(8, txtEndereco.getText());
        stmt.setInt(9, Integer.parseInt(txtNumero.getText()));
        stmt.setString(10, txtComplemento.getText());
        stmt.setString(11, txtBairro.getText());
        stmt.setString(12, txtCidade.getText());
        stmt.setString(13, cbUF.getSelectedItem().toString());
        stmt.setInt(14, Integer.parseInt(txtCodigo.getText()));

        stmt.executeUpdate();
        JOptionPane.showMessageDialog(null, "Cliente atualizado com sucesso!");
        conexao.close();
    } catch (SQLException erro) {
        JOptionPane.showMessageDialog(null, "Erro ao editar: " + erro.getMessage());
    }
}

11 Botão "Excluir" — DELETE

Frmclientes.javaprivate void btnExcluirActionPerformed(java.awt.event.ActionEvent evt) {
    if (txtCodigo.getText().isEmpty()) {
        JOptionPane.showMessageDialog(null, "Selecione um cliente na Consulta antes de excluir.");
        return;
    }

    int confirmacao = JOptionPane.showConfirmDialog(null, "Tem certeza que deseja excluir?");
    if (confirmacao != JOptionPane.YES_OPTION) return;

    Connection conexao = conectar();
    if (conexao == null) return;

    try {
        String sql = "DELETE FROM tb_clientes WHERE ID=?";
        PreparedStatement stmt = conexao.prepareStatement(sql);
        stmt.setInt(1, Integer.parseInt(txtCodigo.getText()));

        stmt.executeUpdate();
        JOptionPane.showMessageDialog(null, "Cliente excluído com sucesso!");
        conexao.close();
    } catch (SQLException erro) {
        JOptionPane.showMessageDialog(null, "Erro ao excluir: " + erro.getMessage());
    }
}
